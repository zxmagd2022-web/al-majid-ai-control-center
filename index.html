async function createSellerStore(){ if(!currentUser || isDirector) return; const name = val("newSellerName"); if(!name){ el("sellerPortalMessage").textContent = "اكتب اسم المتجر."; return; } try{ const {error} = await db .from("am_market_sellers") .insert({ id: currentUser.id, display_name: name, phone: val("newSellerPhone"), owner_user_id: currentUser.id, status: "pending" }); if(error) throw error; el("sellerPortalMessage").textContent = "تم إرسال المتجر للمراجعة بنجاح."; await loadSellerPortal(); }catch(e){ el("sellerPortalMessage").textContent = "تعذر إنشاء المتجر: " + e.message; } }
